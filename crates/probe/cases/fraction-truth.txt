# The contexts of `XOR`, on the string (`NOT (a XOR b)` read as `(NOT a) XOR b` before the fixes).
xor_is_true (SELECT GROUP_CONCAT(id, ':', (s XOR 1) IS TRUE ORDER BY id) FROM ft)
xor_not_not (SELECT GROUP_CONCAT(id, ':', IFNULL(NOT NOT (s XOR 1), 'n') ORDER BY id) FROM ft)
xor_and (SELECT GROUP_CONCAT(id, ':', IFNULL((s XOR 1) AND 1, 'n') ORDER BY id) FROM ft)
xor_if (SELECT GROUP_CONCAT(id, ':', IF(s XOR 1, 1, 0) ORDER BY id) FROM ft)
xor_chain (SELECT GROUP_CONCAT(id, ':', IFNULL(s XOR 1 XOR 0, 'n') ORDER BY id) FROM ft)
not_xor_left (SELECT GROUP_CONCAT(id, ':', IFNULL(NOT (s XOR 1), 'n') ORDER BY id) FROM ft)
not_xor_right (SELECT GROUP_CONCAT(id, ':', IFNULL(NOT (1 XOR s), 'n') ORDER BY id) FROM ft)
not_xor_chain (SELECT GROUP_CONCAT(id, ':', IFNULL(NOT (s XOR 1 XOR 0), 'n') ORDER BY id) FROM ft)
not_xor_bang (SELECT GROUP_CONCAT(id, ':', IFNULL(!(s XOR 1), 'n') ORDER BY id) FROM ft)
not_xor_same (SELECT GROUP_CONCAT(id, ':', IFNULL(NOT (s XOR s), 'n') ORDER BY id) FROM ft)
not_of_not_s (SELECT GROUP_CONCAT(id, ':', IFNULL((NOT s) XOR 1, 'n') ORDER BY id) FROM ft)
where_not_xor (SELECT GROUP_CONCAT(id ORDER BY id) FROM ft WHERE NOT (s XOR 1))
where_xor_is_true (SELECT GROUP_CONCAT(id ORDER BY id) FROM ft WHERE (s XOR 0) IS TRUE)
