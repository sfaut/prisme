# Functions of a string whose other arguments are binary strings, by version.
r_from REPLACE('xaXb', CAST('X' AS BINARY), '-')
r_to REPLACE('xaxb', 'x', CAST('-' AS BINARY))
r_utf8 REPLACE('ébé', CAST('b' AS BINARY), '-')
r_str REPLACE(CAST('xaXb' AS BINARY), 'X', '-')
r_bad REPLACE('ab', CAST(X'FF' AS BINARY), '-')
si_delim SUBSTRING_INDEX('a.b.c', CAST('.' AS BINARY), 2)
si_case SUBSTRING_INDEX('aXbxc', CAST('x' AS BINARY), 1)
si_utf8 SUBSTRING_INDEX('é.b', CAST('.' AS BINARY), 1)
tb_rem TRIM(BOTH CAST('x' AS BINARY) FROM 'xxabxx')
tb_bad TRIM(BOTH CAST(X'FF' AS BINARY) FROM 'ab')
pb_cast POSITION(CAST('B' AS BINARY) IN 'abc')
pb_emul POSITION(CAST('B' AS BINARY) IN CAST('abc' AS BINARY))
pb_bytes_emul POSITION(CAST('b' AS BINARY) IN CAST('éb' AS BINARY))
pb_bad LOCATE(CAST(X'FF' AS BINARY), 'ab')
i_cast INSTR('abc', CAST('B' AS BINARY))
pc_collate POSITION('B' COLLATE utf8mb4_bin IN 'abc')
pc_subject POSITION('B' IN 'abc' COLLATE utf8mb4_bin)
tc_collate TRIM(BOTH 'X' COLLATE utf8mb4_bin FROM 'xxabxx')
# LOCATE from a start: in characters of the string, then in bytes of a binary substring.
l_start LOCATE(CAST('b' AS BINARY), 'ébéb', 3)
l_start_far LOCATE(CAST('b' AS BINARY), 'ééébéb', 4)
l_start_bin LOCATE(CAST('b' AS BINARY), CAST('ébéb' AS BINARY), 3)
l_start_chars LOCATE('b', 'ébéb', 3)
l_beyond LOCATE(CAST('b' AS BINARY), 'éééb', 4)
