# GREATEST and LEAST over mixed types.
# Numbers only
int LEAST(1, 2)
int_big_unsigned LEAST(18446744073709551615, -1)
g_big_unsigned GREATEST(18446744073709551615, -1)
int_dec LEAST(1, 2.5)
dec_dbl LEAST(1.5, 2e0)
int_bool LEAST(TRUE, 2)
int_null LEAST(1, NULL)
null_only LEAST(NULL, NULL)
# Strings and numbers
str_int LEAST('300', 2147483648)
str_int_small LEAST('10', 9)
g_str_int GREATEST('10', 9)
str_dec LEAST('10', 9.5)
str_dbl LEAST('10', 9e0)
alpha_int LEAST('abc', 5)
g_alpha_int GREATEST('abc', 5)
str_str_int LEAST('10', '9', 8)
str_neg LEAST('-5', 3)
str_space LEAST(' 5', 3)
str_exp LEAST('1e3', 999)
str_dec_text LEAST('9.50', 10)
col_str_int LEAST(s, i)
col_int_dec LEAST(i, n)
col_str_dbl LEAST(s, f)
# Strings only
ci LEAST('a', 'B')
bin LEAST(BINARY 'a', 'B')
bin_num LEAST(_binary '10', 9)
hex_num LEAST(0x41, 66)
hex_str LEAST(0x41, 'a')
# Temporal
date_str LEAST(DATE '2024-01-10', '2024-01-9')
g_date_str GREATEST(DATE '2024-01-10', '2024-01-9')
date_bad_str LEAST(DATE '2024-01-10', 'abc')
date_dt LEAST(DATE '2024-01-10', TIMESTAMP '2024-01-09 10:00:00')
date_int LEAST(DATE '2024-01-10', 20240109)
date_dec LEAST(DATE '2024-01-10', 20240109.5)
time_str LEAST(TIME '10:00:00', '9:00:00')
date_time LEAST(DATE '2024-01-10', TIME '10:00:00')
dt_ts LEAST(dt, ts)
d_col_str LEAST(d, s)
d_col_dt LEAST(d, dt)
t_col_int LEAST(t, i)
# JSON
json_int LEAST(j, 1)
json_str LEAST(j, 'a')
