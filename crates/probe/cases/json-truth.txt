# Truth value of a JSON value read as a boolean, by the operator that reads it.
! CREATE TEMPORARY TABLE jt (id INT, v JSON)
! INSERT INTO jt VALUES (1, '{"a": 1}'), (2, '[]'), (3, 'true'), (4, 'false'), (5, '0'), (6, '1'), (7, '"0"'), (8, '"abc"'), (9, 'null'), (10, '0.5'), (11, '"1"')
not_ (SELECT GROUP_CONCAT(id, ':', NOT v ORDER BY id) FROM jt)
and_ (SELECT GROUP_CONCAT(id, ':', v AND 1 ORDER BY id) FROM jt)
or_ (SELECT GROUP_CONCAT(id, ':', v OR 0 ORDER BY id) FROM jt)
xor_ (SELECT GROUP_CONCAT(id, ':', v XOR 0 ORDER BY id) FROM jt)
is_true (SELECT GROUP_CONCAT(id, ':', v IS TRUE ORDER BY id) FROM jt)
is_unknown (SELECT GROUP_CONCAT(id, ':', v IS UNKNOWN ORDER BY id) FROM jt)
if_ (SELECT GROUP_CONCAT(id, ':', IF(v, 1, 0) ORDER BY id) FROM jt)
case_ (SELECT GROUP_CONCAT(id, ':', CASE WHEN v THEN 1 ELSE 0 END ORDER BY id) FROM jt)
where_ (SELECT GROUP_CONCAT(id ORDER BY id) FROM jt WHERE v)
where_not (SELECT GROUP_CONCAT(id ORDER BY id) FROM jt WHERE NOT v)
having_ (SELECT GROUP_CONCAT(id ORDER BY id) FROM (SELECT id FROM jt GROUP BY id, v HAVING v) AS h)
on_ (SELECT GROUP_CONCAT(jt.id ORDER BY jt.id) FROM jt JOIN (SELECT 1 AS x) AS o ON jt.v)
plus_ (SELECT GROUP_CONCAT(id, ':', v + 0 ORDER BY id) FROM jt)
# Its text, read by MySQL as MariaDB reads its JSON, a LONGTEXT.
text (SELECT GROUP_CONCAT(id, ':', CAST(v AS CHAR) ORDER BY id SEPARATOR ' | ') FROM jt)
text_not_ (SELECT GROUP_CONCAT(id, ':', NOT CAST(v AS CHAR) ORDER BY id) FROM jt)
text_and_ (SELECT GROUP_CONCAT(id, ':', CAST(v AS CHAR) AND 1 ORDER BY id) FROM jt)
text_xor_ (SELECT GROUP_CONCAT(id, ':', CAST(v AS CHAR) XOR 0 ORDER BY id) FROM jt)
text_is_true (SELECT GROUP_CONCAT(id, ':', CAST(v AS CHAR) IS TRUE ORDER BY id) FROM jt)
text_if_ (SELECT GROUP_CONCAT(id, ':', IF(CAST(v AS CHAR), 1, 0) ORDER BY id) FROM jt)
text_case_ (SELECT GROUP_CONCAT(id, ':', CASE WHEN CAST(v AS CHAR) THEN 1 ELSE 0 END ORDER BY id) FROM jt)
text_where_ (SELECT GROUP_CONCAT(id ORDER BY id) FROM jt WHERE CAST(v AS CHAR))
text_where_not (SELECT GROUP_CONCAT(id ORDER BY id) FROM jt WHERE NOT CAST(v AS CHAR))
