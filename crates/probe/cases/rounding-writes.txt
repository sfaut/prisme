# Forms written towards MySQL for a temporal argument of MariaDB: an integer
# for a date, a decimal of the digits of the argument for ABS.
w_ceil_d CEIL(d + 0)
w_floor_d FLOOR(d + 0)
w_round_d ROUND(d + 0)
w_round_d_m1 ROUND(d + 0, -1)
w_round_d_1 ROUND(d + 0, 1)
w_truncate_d_m1 TRUNCATE(d + 0, -1)
w_truncate_d_1 TRUNCATE(d + 0, 1)
w_ceil_dn CEIL(dn + 0)
w_abs_d ABS(CAST(d AS DECIMAL(8)))
w_abs_dn ABS(CAST(dn AS DECIMAL(8)))
w_abs_dt ABS(CAST(dt AS DECIMAL(17, 3)))
w_abs_ts ABS(CAST(ts AS DECIMAL(20, 6)))
w_abs_t ABS(CAST(t AS DECIMAL(7)))
abs_tneg ABS(TIME '-10:20:30.5')
w_abs_tneg ABS(CAST(TIME '-10:20:30.5' AS DECIMAL(8, 1)))
abs_tmax ABS(TIME '-838:59:59.999999')
w_abs_tmax ABS(CAST(TIME '-838:59:59.999999' AS DECIMAL(13, 6)))
abs_dtmax ABS(TIMESTAMP '9999-12-31 23:59:59.999999')
w_abs_dtmax ABS(CAST(TIMESTAMP '9999-12-31 23:59:59.999999' AS DECIMAL(20, 6)))
abs_dmax ABS(DATE '9999-12-31')
w_abs_dmax ABS(CAST(DATE '9999-12-31' AS DECIMAL(8)))
ceil_dmax CEIL(DATE '9999-12-31')
w_ceil_dmax CEIL(DATE '9999-12-31' + 0)
round_dmax ROUND(DATE '9999-12-31', -4)
w_round_dmax ROUND(DATE '9999-12-31' + 0, -4)
ceil_tneg CEIL(TIME '-10:20:30.5')
floor_tneg FLOOR(TIME '-10:20:30.5')
round_tneg ROUND(TIME '-10:20:30.5')
abs_e_tneg ABS(TIME '-10:20:30.5' + 0e0)
round_d_m9 ROUND(d, -9)
w_round_d_m9 ROUND(d + 0, -9)
truncate_d_m9 TRUNCATE(d, -9)
w_truncate_d_m9 TRUNCATE(d + 0, -9)
round_d_big ROUND(d, 40)
w_round_d_big ROUND(d + 0, 40)
