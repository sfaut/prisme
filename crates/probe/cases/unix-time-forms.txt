# Zero TIMESTAMP, epoch rebuilt, forms of the emulations, in UTC
! SET time_zone = '+00:00'
! SET SESSION sql_mode = ''
! DROP TABLE IF EXISTS z
! CREATE TABLE z (ts TIMESTAMP NULL, dt DATETIME NULL, n BIGINT NULL, u INT UNSIGNED NULL)
! INSERT INTO z VALUES ('0000-00-00 00:00:00', '0000-00-00 00:00:00', 0, 4294967295)
zero_ts        (SELECT UNIX_TIMESTAMP(ts) FROM z)
zero_dt        (SELECT UNIX_TIMESTAMP(dt) FROM z)
epoch_minus    FROM_UNIXTIME(1) - INTERVAL 1 SECOND
epoch_cast     CAST(FROM_UNIXTIME(1) - INTERVAL 1 SECOND AS DATETIME)
epoch_tz       CONVERT_TZ('1970-01-01 00:00:00', '+00:00', @@session.time_zone)
case_cap       CASE WHEN UNIX_TIMESTAMP('2050-01-01') < 2147483648 THEN UNIX_TIMESTAMP('2050-01-01') WHEN '2050-01-01' IS NOT NULL THEN 0 END
case_frac      CASE WHEN UNIX_TIMESTAMP(dt) < 2147483648 THEN UNIX_TIMESTAMP(dt) ELSE 0 END
case_null      CASE WHEN UNIX_TIMESTAMP(dn) < 2147483648 THEN UNIX_TIMESTAMP(dn) WHEN dn IS NOT NULL THEN 0 END
ifnull_frac    IFNULL(UNIX_TIMESTAMP('1960-01-01 00:00:00.5'), 0)
nullif_text    FROM_UNIXTIME(NULLIF('0', 0))
nullif_frac    FROM_UNIXTIME(NULLIF(0.5, 0))
if_text        FROM_UNIXTIME(IF('1700000000' < 2147483648, '1700000000', NULL))
if_frac        FROM_UNIXTIME(IF(1.5 < 2147483648, 1.5, NULL))
cast_text      CAST(FROM_UNIXTIME('1700000000') AS DATETIME(6))
cast_dec       CAST(FROM_UNIXTIME(1.25) AS DATETIME(2))
fmt_epoch      DATE_FORMAT(FROM_UNIXTIME(1) - INTERVAL 1 SECOND, '%Y %H')
