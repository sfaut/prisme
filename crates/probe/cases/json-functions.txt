# The functions of JSON of MySQL: the type of their result, then a `JSON` they give in `IN` and
# read as a boolean, as a `JSON` column, around 8.0.17 (json-text.txt, json-truth.txt).
! DROP TABLE IF EXISTS jz
! CREATE TABLE jz (id INT, v JSON)
! INSERT INTO jz VALUES (1, '{"a": "b"}'), (2, '{"a": 1}'), (3, '{"a": true}'), (4, '{"a": 0.5}'), (5, '{"a": "1"}'), (6, '{"a": 0}'), (7, '{}'), (8, '{"a": [0]}')
# A `JSON`, nullable.
array JSON_ARRAY(1, 'a')
array_empty JSON_ARRAY()
object JSON_OBJECT('k', 1)
extract JSON_EXTRACT('{"a": 1}', '$.a')
array_append JSON_ARRAY_APPEND('[1]', '$', 2)
array_insert JSON_ARRAY_INSERT('[1]', '$[0]', 2)
insert JSON_INSERT('{}', '$.b', 1)
replace JSON_REPLACE('{"b": 0}', '$.b', 1)
set JSON_SET('{}', '$.b', 1)
remove JSON_REMOVE('{"b": 0}', '$.b')
keys JSON_KEYS('{"b": 0}')
merge JSON_MERGE('[1]', '[2]')
merge_patch JSON_MERGE_PATCH('{"a": 1}', '{"b": 2}')
merge_preserve JSON_MERGE_PRESERVE('[1]', '[2]')
search JSON_SEARCH('["x"]', 'one', 'x')
validation_report JSON_SCHEMA_VALIDATION_REPORT('{}', '1')
# A number or a string.
quote JSON_QUOTE('a')
unquote JSON_UNQUOTE('"a"')
type JSON_TYPE('1')
pretty JSON_PRETTY('[1]')
contains JSON_CONTAINS('[1]', '1')
length JSON_LENGTH('[1]')
overlaps JSON_OVERLAPS('[1]', '[1]')
schema_valid JSON_SCHEMA_VALID('{}', '1')
# `IN`: by the rules of JSON from 8.0.17, as `=` and `OR` in every version.
raw_in (SELECT GROUP_CONCAT(id, ':', IFNULL(JSON_EXTRACT(v, '$.a') IN ('b', 1), 'n') ORDER BY id) FROM jz)
or_in (SELECT GROUP_CONCAT(id, ':', IFNULL(JSON_EXTRACT(v, '$.a') = 'b' OR JSON_EXTRACT(v, '$.a') = 1, 'n') ORDER BY id) FROM jz)
array_in (SELECT GROUP_CONCAT(id, ':', JSON_ARRAY(id) IN ('[1]', 1) ORDER BY id) FROM jz)
# Read as a boolean: true unless it is the number 0 from 8.0.17, a number before.
raw_where (SELECT GROUP_CONCAT(id ORDER BY id) FROM jz WHERE JSON_EXTRACT(v, '$.a'))
raw_not (SELECT GROUP_CONCAT(id, ':', IFNULL(NOT JSON_EXTRACT(v, '$.a'), 'n') ORDER BY id) FROM jz)
raw_if (SELECT GROUP_CONCAT(id, ':', IF(JSON_EXTRACT(v, '$.a'), 1, 0) ORDER BY id) FROM jz)
