# Arguments that functions of the same name read otherwise in MySQL and
# MariaDB (rule `function.argument`, `same_except` in the reference files):
# a date or a time in STRCMP; a floating point number in HEX, rounded half to
# even by MySQL, half away from zero by MariaDB; a DECIMAL of more than 15
# digits in HEX, read as a DOUBLE by MariaDB, bounded to the range of BIGINT
# by MySQL. BIN and CONV of the same numbers, alike, for comparison.

strcmp_d_str STRCMP(d, '2024')

strcmp_i_str STRCMP(i, '5')
strcmp_dt_s STRCMP(dt, s)
strcmp_t_s STRCMP(t, s)
strcmp_t_t STRCMP(t, t)
strcmp_d_d STRCMP(d, '2024-05-06')
strcmp_t_time STRCMP(t, '10:20:30')
strcmp_dt_str STRCMP(dt, '2024-05-06 07:08:09.123')
strcmp_t_9 STRCMP(t, '9')
strcmp_n_str STRCMP(n, '12.50')
strcmp_f_str STRCMP(f, '2.5')
