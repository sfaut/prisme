# GREATEST and LEAST: result types.
! DROP TABLE IF EXISTS extra
! CREATE TABLE extra (y YEAR NOT NULL, b BIT(8) NOT NULL, e ENUM('b', 'a') NOT NULL, st SET('x', 'y') NOT NULL, sb VARCHAR(10) COLLATE utf8mb4_bin NOT NULL, vb VARBINARY(10) NOT NULL, t3 TIME(3) NOT NULL, u INT UNSIGNED NOT NULL)
! INSERT INTO extra VALUES (2001, 66, 'a', 'y', 'B', 'A', '-01:02:03.456', 7)
year_int (SELECT LEAST(y, 2000) FROM extra)
year_str (SELECT LEAST(y, '999') FROM extra)
year_dec (SELECT LEAST(y, 2000.5) FROM extra)
year_date (SELECT LEAST(y, d) FROM extra)
bit_int (SELECT LEAST(b, 65) FROM extra)
bit_str (SELECT LEAST(b, 'A') FROM extra)
bit_bit (SELECT LEAST(b, b) FROM extra)
enum_int (SELECT LEAST(e, 1) FROM extra)
enum_str (SELECT LEAST(e, 'aa') FROM extra)
enum_enum (SELECT LEAST(e, e) FROM extra)
set_int (SELECT LEAST(st, 1) FROM extra)
set_str (SELECT LEAST(st, 'x') FROM extra)
unsigned_null (SELECT LEAST(u, NULL) FROM extra)
unsigned_if (SELECT LEAST(u, IF(FALSE, u, NULL)) FROM extra)
dec_if LEAST(n, IF(FALSE, n, NULL))
# Collations: a column of another collation with a number
sb_int (SELECT LEAST(sb, 5) FROM extra)
sb_concat (SELECT LEAST(sb, CONCAT(5)) FROM extra)
sb_cast (SELECT LEAST(sb, CAST(5 AS CHAR)) FROM extra)
sb_lower_int (SELECT GREATEST(sb, 5, 'a') FROM extra)
sb_lower_concat (SELECT GREATEST(sb, CONCAT(5), 'a') FROM extra)
vb_int (SELECT LEAST(vb, 5) FROM extra)
vb_concat (SELECT LEAST(vb, CONCAT(5)) FROM extra)
# Strings to double for MySQL
vb_plus (SELECT LEAST(vb + 0e0, 5) FROM extra)
enum_plus (SELECT LEAST(e + 0e0, 1) FROM extra)
# Fractions of temporal values and strings
dt_str6 LEAST(dt, '2000-01-01 00:00:00.123456')
g_dt_str6 GREATEST(dt, '2030-01-01 00:00:00.123456')
d_str_frac LEAST(d, '2000-01-01 00:00:00.5')
t3_str (SELECT LEAST(t3, '-1:02:03.4567') FROM extra)
t3_str_g (SELECT GREATEST(t3, '-1:02:03.4567') FROM extra)
t3_cast (SELECT LEAST(t3, CAST('-1:02:03.4567' AS TIME(4))) FROM extra)
t3_concat (SELECT LEAST(CONCAT(t3), '-1:02:03.4567') FROM extra)
d_dt_str LEAST(d, dt, '2024-05-06 03:00:00')
date_cast_str LEAST(d, CAST('2024-01-09' AS DATE))
dt_cast_str6 LEAST(dt, CAST('2000-01-01 00:00:00.123456' AS DATETIME(6)))
cast_least_date CAST(LEAST(d, '2024-01-09') AS DATE)
cast_least_dt CAST(LEAST(dt, '2030-01-01 00:00:00.5') AS DATETIME(3))
concat_least_d CONCAT(LEAST(d, '2024-01-9'))
concat_least_dt CONCAT(LEAST(dt, '2030-01-01'))
concat_least_ts CONCAT(LEAST(ts, '2030-01-01'))
