# A JSON of MySQL compared, sorted and converted (raw_), against its text by CONCAT (text_),
# which MariaDB reads from its JSON, a LONGTEXT of collation utf8mb4_bin.
! DROP TABLE IF EXISTS jx
! CREATE TABLE jx (id INT, v JSON)
! INSERT INTO jx VALUES (1, '{"a": 1}'), (2, '[1, 2]'), (3, 'true'), (4, '1'), (5, '"abc"'), (6, '10'), (7, '"b"'), (8, 'null'), (9, '2.5'), (10, NULL), (11, '1.0'), (12, '"1"'), (13, '"B"')
# Collation and coercibility.
coll_v (SELECT CONCAT(COLLATION(v), ' ', COERCIBILITY(v)) FROM jx WHERE id = 1)
coll_concat (SELECT CONCAT(COLLATION(CONCAT(v)), ' ', COERCIBILITY(CONCAT(v))) FROM jx WHERE id = 1)
coll_cast (SELECT CONCAT(COLLATION(CAST(v AS CHAR)), ' ', COERCIBILITY(CAST(v AS CHAR))) FROM jx WHERE id = 1)
# Comparisons and conversions.
raw_eq_text (SELECT GROUP_CONCAT(id, ':', IFNULL(v = 'b', 'n'), IFNULL(v = '"b"', 'n'), IFNULL(v = '{"a": 1}', 'n') ORDER BY id) FROM jx)
text_eq_text (SELECT GROUP_CONCAT(id, ':', IFNULL(CONCAT(v) = 'b', 'n'), IFNULL(CONCAT(v) = '"b"', 'n'), IFNULL(CONCAT(v) = '{"a": 1}', 'n') ORDER BY id) FROM jx)
raw_eq_num (SELECT GROUP_CONCAT(id, ':', IFNULL(v = 1, 'n'), IFNULL(v = 2.5, 'n'), IFNULL(v < 5, 'n') ORDER BY id) FROM jx)
text_eq_num (SELECT GROUP_CONCAT(id, ':', IFNULL(CONCAT(v) = 1, 'n'), IFNULL(CONCAT(v) = 2.5, 'n'), IFNULL(CONCAT(v) < 5, 'n') ORDER BY id) FROM jx)
raw_lt_text (SELECT GROUP_CONCAT(id, ':', IFNULL(v < 'b', 'n') ORDER BY id) FROM jx)
text_lt_text (SELECT GROUP_CONCAT(id, ':', IFNULL(CONCAT(v) < 'b', 'n') ORDER BY id) FROM jx)
raw_between (SELECT GROUP_CONCAT(id, ':', IFNULL(v BETWEEN '1' AND '9', 'n') ORDER BY id) FROM jx)
text_between (SELECT GROUP_CONCAT(id, ':', IFNULL(CONCAT(v) BETWEEN '1' AND '9', 'n') ORDER BY id) FROM jx)
raw_case (SELECT GROUP_CONCAT(id, ':', CASE v WHEN 'b' THEN 'b' WHEN 1 THEN 'one' ELSE '-' END ORDER BY id) FROM jx)
text_case (SELECT GROUP_CONCAT(id, ':', CASE CONCAT(v) WHEN 'b' THEN 'b' WHEN 1 THEN 'one' ELSE '-' END ORDER BY id) FROM jx)
raw_nullsafe (SELECT GROUP_CONCAT(id, ':', v <=> '"b"' ORDER BY id) FROM jx)
text_nullsafe (SELECT GROUP_CONCAT(id, ':', CONCAT(v) <=> '"b"' ORDER BY id) FROM jx)
raw_plus (SELECT GROUP_CONCAT(id, ':', IFNULL(v + 0, 'n') ORDER BY id) FROM jx)
text_plus (SELECT GROUP_CONCAT(id, ':', IFNULL(CONCAT(v) + 0, 'n') ORDER BY id) FROM jx)
raw_cast (SELECT GROUP_CONCAT(id, ':', IFNULL(CAST(v AS SIGNED), 'n'), '/', IFNULL(CAST(v AS DECIMAL(5, 1)), 'n') ORDER BY id) FROM jx)
text_cast (SELECT GROUP_CONCAT(id, ':', IFNULL(CAST(CONCAT(v) AS SIGNED), 'n'), '/', IFNULL(CAST(CONCAT(v) AS DECIMAL(5, 1)), 'n') ORDER BY id) FROM jx)
raw_in (SELECT GROUP_CONCAT(id, ':', IFNULL(v IN ('b', 1), 'n') ORDER BY id) FROM jx)
text_in (SELECT GROUP_CONCAT(id, ':', IFNULL(CONCAT(v) IN ('b', 1), 'n') ORDER BY id) FROM jx)
# Sorts, groups, aggregates.
raw_order (SELECT GROUP_CONCAT(id ORDER BY v, id) FROM jx)
text_order (SELECT GROUP_CONCAT(id ORDER BY CONCAT(v), id) FROM jx)
raw_distinct (SELECT COUNT(*) FROM (SELECT DISTINCT v FROM jx) AS q)
text_distinct (SELECT COUNT(*) FROM (SELECT DISTINCT CONCAT(v) AS v FROM jx ORDER BY CONCAT(v)) AS q)
raw_count_distinct (SELECT COUNT(DISTINCT v) FROM jx)
text_count_distinct (SELECT COUNT(DISTINCT CONCAT(v)) FROM jx)
raw_group (SELECT GROUP_CONCAT(n ORDER BY n) FROM (SELECT v, COUNT(*) AS n FROM jx GROUP BY v) AS q)
text_group (SELECT GROUP_CONCAT(n ORDER BY n) FROM (SELECT CONCAT(v) AS v, COUNT(*) AS n FROM jx GROUP BY CONCAT(v)) AS q)
raw_having (SELECT GROUP_CONCAT(n ORDER BY n) FROM (SELECT COUNT(*) AS n FROM jx GROUP BY v HAVING v < '2') AS q)
text_having (SELECT GROUP_CONCAT(n ORDER BY n) FROM (SELECT COUNT(*) AS n FROM jx GROUP BY CONCAT(v), v HAVING CONCAT(v) < '2') AS q)
text_having_alone (SELECT GROUP_CONCAT(n ORDER BY n) FROM (SELECT COUNT(*) AS n FROM jx GROUP BY CONCAT(v) HAVING CONCAT(v) < '2') AS q)
raw_union (SELECT COUNT(*) FROM (SELECT v FROM jx UNION SELECT v FROM jx) AS q)
text_union (SELECT COUNT(*) FROM (SELECT CONCAT(v) AS v FROM jx UNION SELECT CONCAT(v) FROM jx) AS q)
text_union_order (SELECT GROUP_CONCAT(v SEPARATOR ' | ') FROM ((SELECT v FROM jx WHERE id < 5) UNION ALL (SELECT v FROM jx WHERE id > 10) ORDER BY CONCAT(v)) AS q)
raw_partition (SELECT GROUP_CONCAT(id, ':', n ORDER BY id) FROM (SELECT id, COUNT(*) OVER (PARTITION BY v) AS n FROM jx) AS q)
text_partition (SELECT GROUP_CONCAT(id, ':', n ORDER BY id) FROM (SELECT id, COUNT(*) OVER (PARTITION BY CONCAT(v)) AS n FROM jx) AS q)
raw_min_max (SELECT CONCAT(MIN(v), ' | ', MAX(v), ' | ', SUM(v)) FROM jx)
text_min_max (SELECT CONCAT(MIN(CONCAT(v)), ' | ', MAX(CONCAT(v)), ' | ', SUM(CONCAT(v))) FROM jx)
raw_join (SELECT COUNT(*) FROM jx AS a JOIN jx AS b ON a.v = b.v AND a.id <> b.id)
text_join (SELECT COUNT(*) FROM jx AS a JOIN jx AS b ON CONCAT(a.v) = CONCAT(b.v) AND a.id <> b.id)
raw_in_subquery (SELECT GROUP_CONCAT(o.id ORDER BY o.id) FROM jx AS o WHERE 'b' IN (SELECT i.v FROM jx AS i WHERE i.id = o.id))
text_in_subquery (SELECT GROUP_CONCAT(o.id ORDER BY o.id) FROM jx AS o WHERE 'b' IN (SELECT CONCAT(i.v) FROM jx AS i WHERE i.id = o.id))
raw_scalar (SELECT GROUP_CONCAT(o.id ORDER BY o.id) FROM jx AS o WHERE (SELECT i.v FROM jx AS i WHERE i.id = o.id) = 'b')
text_scalar (SELECT GROUP_CONCAT(o.id ORDER BY o.id) FROM jx AS o WHERE CONCAT((SELECT i.v FROM jx AS i WHERE i.id = o.id)) = 'b')
# Read alike: its text.
like_ (SELECT GROUP_CONCAT(id, ':', IFNULL(v LIKE '%b%', 'n'), IFNULL(v LIKE '%B%', 'n') ORDER BY id) FROM jx)
concat_ (SELECT GROUP_CONCAT(id, ':', CONCAT(v, '|'), LENGTH(v) ORDER BY id SEPARATOR ' ') FROM jx)
cast_char (SELECT GROUP_CONCAT(id, ':', IFNULL(CAST(v AS CHAR), 'n') ORDER BY id SEPARATOR ' ') FROM jx)
count_ (SELECT CONCAT(COUNT(v), ' ', SUM(v IS NULL)) FROM jx)
# The text stored: normalized by MySQL, as written by MariaDB.

! CREATE TABLE jy (v JSON)

stored (SELECT GROUP_CONCAT(v SEPARATOR ' | ') FROM jy)
